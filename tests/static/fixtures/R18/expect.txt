# R18 견본별 기대값 — 형식: <견본 폴더> rc=<종료 코드> findings=<FINDING 줄 수>
# 부르는 법: bash tests/static/selftest.sh R18
#   (견본 폴더는 git 작업 트리의 맨 위가 아니라서 디렉터리 전체의 scripts·tests 를 본다)
# 견본은 전부 새로 지은 가짜 코드다(정답지·저장소 본문 복사 없음, 주소는 example.invalid).
# 견본 PHP 는 `.php.in` 으로 둔다(SQL·쓰기 모양이 저장소에서 scripts/security_lint.sh 에 걸리지 않게).
#   fire_php_flag       깃발 흐름 — `if ($a || $b)` 안의 상태 저장, $b 를 켜는 자리에 dry-run 조건 없음(1)
#   fire_php_direct     맨 위의 file_put_contents · 막힌 unlink/rename/삼항·|| 는 조용(1)
#   fire_php_lib_helper require 한 파일의 쓰기 도우미를 dry-run 조건 없이 부름 — 부르는 줄(1)
#   fire_sh_direct      셸 `>` 리다이렉트(1) · 막힌 cp/rm/tee, mktemp·>&2 는 조용
#   fire_sh_helper      같은 파일 셸 함수의 쓰기 — 함수 안 쓰기 줄(1) · dry-run 을 보는 run() 감싸기는 조용
#   fire_sh_sql         "$1" 을 psql 에 넘기는 도우미로 UPDATE — dry 종료보다 앞(1)
#   fire_py             파이썬 Path.write_text(1) · not args.dry_run·앞선 return 은 조용
#   quiet_guarded       막는 모양 모음(앞선 종료·else·elseif·깃발 흐름·클로저 return·잠금·임시) — 0
#   quiet_not_dryrun    다른 스크립트에 --dry-run 을 넘기기만 하는 시험(쓰기 있음) — 대상 밖 0
fire_php_flag rc=1 findings=1
fire_php_direct rc=1 findings=1
fire_php_lib_helper rc=1 findings=1
fire_sh_direct rc=1 findings=1
fire_sh_helper rc=1 findings=1
fire_sh_sql rc=1 findings=1
fire_py rc=1 findings=1
quiet_guarded rc=0 findings=0
quiet_not_dryrun rc=0 findings=0
