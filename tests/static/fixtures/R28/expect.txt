#   fire_lookup      set -euo pipefail 아래 id -u "$U" · getent … | cut(pipefail) · readlink -e 대입(3) — HO-P7-007 모양
#   fire_shebang     셔뱅 `bash -e` 아래 pgrep · command -v 대입(2)
#   fire_outer       바깥 반영 스크립트(--outer)의 set -e + id -u "$OWNER"(1)
#   quiet_ok         || { … } · if ! · 앞선 같은 조회 확인 · local · id -u(자기) · set +e 구간 · readlink -f ·
#                    치환 안 || true · pipefail 없이 파이프 앞자리 · set -e 없는 파일 · 표식 — 0

fire_outer rc=1 findings=1
